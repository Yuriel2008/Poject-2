        // Data is now structured in levels, each with 3 pairs
        const levels = [
            [
                { id: 1, meaning: 'to gain knowledge', word: 'học' },
                { id: 2, meaning: 'a book', word: 'quyển sách' },
                { id: 3, meaning: 'to eat', word: 'ăn' }
            ],
            [
                { id: 4, meaning: 'water', word: 'nước' },
                { id: 5, meaning: 'hello', word: 'xin chào' },
                { id: 6, meaning: 'thank you', word: 'cảm ơn' }
            ],
            [
                { id: 7, meaning: 'goodbye', word: 'tạm biệt' },
                { id: 8, meaning: 'house', word: 'nhà' },
                { id: 9, meaning: 'teacher', word: 'giáo viên' }
            ]
        ];
        let currentLevelIndex = 0;
        let selectedItem = null;
        let matchesFound = 0;
        const meaningsColumn = document.getElementById('meanings-column');
        const wordsColumn = document.getElementById('words-column');
        const nextLevelBtn = document.getElementById('next-level-btn');
        const menu = document.getElementById('tools-menu');
        const menuIcon = document.getElementById('menu-icon');
        // Utility to shuffle an array
        function shuffle(array) {
            return array.sort(() => Math.random() - 0.5);
        }
        function loadLevel(levelIndex) {
            matchesFound = 0;
            const levelData = levels[levelIndex];
            // Clear previous items and hide button
            meaningsColumn.innerHTML = '';
            wordsColumn.innerHTML = '';
            nextLevelBtn.classList.remove('show');
            // Separate and shuffle meanings and words
            const meanings = shuffle(levelData.map(item => ({ text: item.meaning, id: item.id })));
            const words = shuffle(levelData.map(item => ({ text: item.word, id: item.id })));
            // Create and append items to columns
            meanings.forEach(item => createMatchItem(item, meaningsColumn));
            words.forEach(item => createMatchItem(item, wordsColumn));
        }
        function createMatchItem(itemData, column) {
            const item = document.createElement('div');
            item.className = 'match-item';
            item.textContent = itemData.text;
            item.dataset.id = itemData.id;
            item.dataset.column = column.id;
            item.addEventListener('click', () => onItemClick(item));
            column.appendChild(item);
        }
        function onItemClick(item) {
            if (item.classList.contains('correct')) return; // Already matched
            if (!selectedItem) {
                // First selection
                selectedItem = item;
                item.classList.add('selected');
            } else {
                // Second selection
                if (selectedItem === item || selectedItem.dataset.column === item.dataset.column) {
                    // Deselect if clicking the same item or an item in the same column
                    selectedItem.classList.remove('selected');
                    selectedItem = null;
                    return;
                }
                // Check for a match
                if (selectedItem.dataset.id === item.dataset.id) {
                    // It's a match!
                    selectedItem.classList.add('correct');
                    item.classList.add('correct');
                    selectedItem.classList.remove('selected');
                    matchesFound++;
                    selectedItem = null;
                    if (matchesFound === 3) {
                        nextLevelBtn.classList.add('show');
                    }
                } else {
                    // It's not a match
                    selectedItem.classList.add('incorrect');
                    item.classList.add('incorrect');
                    // Reset after a short delay
                    setTimeout(() => {
                        selectedItem.classList.remove('incorrect', 'selected');
                        item.classList.remove('incorrect');
                        selectedItem = null;
                    }, 500);
                }
            }
        }
        function nextLevel() {
            currentLevelIndex = (currentLevelIndex + 1) % levels.length;
            loadLevel(currentLevelIndex);
        }
        function toggleMenu() {
            menu.classList.toggle('show');
        }
        window.onclick = function(event) {
            if (!menu.contains(event.target) && !menuIcon.contains(event.target)) {
                if (menu.classList.contains('show')) {
                    menu.classList.remove('show');
                }
            }
        }
        // Initialize first level on page load
        window.onload = () => {
            loadLevel(currentLevelIndex);
            // You can re-add the menu tools HTML here if needed
            document.getElementById('tools-menu').innerHTML = `
                <div class="tool-card progress-tracker">
                    <h3>Progress Tracker</h3>
                    <ul>
                        <li>Levels Completed: <span id="levels-completed">0</span></li>
                    </ul>
                </div>
                <div class="tool-card habit-tracker">
                    <h3>Habit Tracker</h3>
                    <ul>

                    </ul>
                </div>
                <div class="tool-card">
                    <h3>Quick Notes</h3>
                    <textarea id="quick-notes-textarea" placeholder="Write down new words or phrases..."></textarea>
                </div>`;
        };
    </script>

                this._bindEvents();
                this._loadNextRound();

            _bindEvents() {
                this.elements.difficultySelector.addEventListener('click', this._handleDifficultyChange.bind(this));
                this.elements.nextLevelBtn.addEventListener('click', this._loadNextRound.bind(this));
                this.elements.menuIcon.addEventListener('click', this._toggleMenu.bind(this));
                this.elements.closeArchiveBtn.addEventListener('click', this._closeArchive.bind(this));
                this.elements.wordsColumn.addEventListener('click', this._handleItemClick.bind(this));
                this.elements.meaningsColumn.addEventListener('click', this._handleItemClick.bind(this));
                window.addEventListener('click', (event) => {
                    if (event.target === this.elements.archiveModal) this._closeArchive();
                    if (this.elements.menu.classList.contains('show') && !this.elements.menu.contains(event.target) && !this.elements.menuIcon.contains(event.target)) {
                        this.elements.menu.classList.remove('show');
                    }
                });
            },
            /**
             * Loads the next round of UNSEEN words for the current difficulty.
             */
            _loadNextRound() {
                this.state.matchesFound = 0;
                this.state.selectedItem = null;
                // 1. Get the pool of all words for the current difficulty.
                const wordPool = this.gameData[this.state.currentDifficulty];
                // 2. Filter out words that have already been mastered in this session.
                const unseenWords = wordPool.filter(word => !this.state.masteredWordIds.has(word.id));
                // 3. Check for completion.
                if (unseenWords.length === 0) {
                    this._updateFeedback(`Difficulty Complete! Try the next level.`);
                    this.elements.wordsColumn.innerHTML = '';
                    this.elements.meaningsColumn.innerHTML = '';
                    this.elements.nextLevelBtn.classList.remove('show');
                    return;
                }
                // 4. Get the next 3 words and render them.
                this.state.currentLevelData = this._shuffle(unseenWords).slice(0, 3);
                this._renderLevel();
            },
            /**
             * Renders the UI for the current round of words.
             */
            _renderLevel() {
                this.elements.wordsColumn.innerHTML = '';
                this.elements.meaningsColumn.innerHTML = '';
                this._updateFeedback('');
                this.elements.nextLevelBtn.classList.remove('show');
                const createMatchItem = (itemData) => {
                    const item = document.createElement('div');
                    item.className = 'match-item';
                    item.textContent = itemData.text;
                    item.dataset.id = itemData.id;
                    return item;
                };
                const words = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.word, id: item.id })));
                const meanings = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.meaning, id: item.id })));
                words.forEach(item => this.elements.wordsColumn.appendChild(createMatchItem(item)));
                meanings.forEach(item => this.elements.meaningsColumn.appendChild(createMatchItem(item)));
            },
            _handleDifficultyChange(event) {
                const target = event.target.closest('.difficulty-btn');
                if (!target) return;
                this.state.currentDifficulty = target.dataset.difficulty;
                // **CRITICAL:** Reset progress when changing difficulty.
                this.state.masteredWordIds.clear();
                document.querySelectorAll('.difficulty-btn').forEach(btn => btn.classList.remove('active'));
                target.classList.add('active');
                this._loadNextRound();
            },
            _handleItemClick(event) {
                const item = event.target.closest('.match-item');
                if (!item || item.classList.contains('correct')) return;
                if (!this.state.selectedItem) {
                    this.state.selectedItem = item;
                    item.classList.add('selected');
                } else {
                    if (this.state.selectedItem === item || this.state.selectedItem.parentElement === item.parentElement) {
                        this.state.selectedItem.classList.remove('selected');
                        this.state.selectedItem = null;
                        return;
                    }
                    if (this.state.selectedItem.dataset.id === item.dataset.id) {
                        const matchedId = parseInt(item.dataset.id, 10);
                        // Add to mastered set so it won't be picked again.
                        this.state.masteredWordIds.add(matchedId);
                        this.state.selectedItem.classList.add('correct');
                        item.classList.add('correct');
                        this.state.matchesFound++;
                        this.state.selectedItem = null;
                        if (this.state.matchesFound === this.state.currentLevelData.length) {
                            this._updateFeedback('Round Complete!', true);
                            this.elements.nextLevelBtn.classList.add('show');
                        }
                    } else {
                        this.state.selectedItem.classList.add('incorrect');
                        item.classList.add('incorrect');
                        const selected = this.state.selectedItem;
                        setTimeout(() => {
                            selected.classList.remove('incorrect', 'selected');
                            item.classList.remove('incorrect');
                        }, 500);
                        this.state.selectedItem = null;
                    }
                }
            },
            _updateFeedback(message, isSuccess = false) {
                this.elements.feedback.textContent = message;
                this.elements.feedback.style.color = isSuccess ? 'var(--success-color)' : 'var(--light-text-color)';
            },
            _shuffle: (array) => array.sort(() => Math.random() - 0.5),
            // --- Menu and Modal Methods (Unchanged) ---

        function onItemClick(item) {
            if (item.classList.contains('correct') || !item.parentElement.childElementCount) return;
            if (!selectedItem) {
                selectedItem = item;
                item.classList.add('selected');
            } else {
                if (selectedItem === item || selectedItem.dataset.column === item.dataset.column) {
                    selectedItem.classList.remove('selected');
                    selectedItem = null;
                    return;
                }
                if (selectedItem.dataset.id === item.dataset.id) { // Match

                    item.classList.add('correct');
                    selectedItem.classList.remove('selected');

                    completedWordIds.add(parseInt(item.dataset.id));

                    selectedItem = null;

                masterList.forEach(word => this.state.wordBank.set(word.id, {...word, lastSeen: 0}));

                this._bindEvents();
                // 4. Load the initial level
                this._loadLevel();

             * Centralized event binding.

                this.elements.difficultySelector.addEventListener('click', this._handleDifficultyChange.bind(this));
                this.elements.nextLevelBtn.addEventListener('click', this._handleNextLevel.bind(this));
                this.elements.menuIcon.addEventListener('click', this._toggleMenu.bind(this));
                this.elements.closeArchiveBtn.addEventListener('click', this._closeArchive.bind(this));
                // Use event delegation for dynamically created match items
                this.elements.wordsColumn.addEventListener('click', this._handleItemClick.bind(this));
                this.elements.meaningsColumn.addEventListener('click', this._handleItemClick.bind(this));
                // Global event listeners
                window.addEventListener('click', (event) => {
                    if (event.target === this.elements.archiveModal) this._closeArchive();
                    if (!this.elements.menu.contains(event.target) && !this.elements.menuIcon.contains(event.target)) {
                        this.elements.menu.classList.remove('show');
                    }
                });

             * Loads a new level based on the current state (difficulty and level index).
             */

                this.state.matchesFound = 0;
                this.state.selectedItem = null;
                if (this.state.currentDifficulty === 'review') {
                    this.state.currentLevelData = this._generateSmartReviewData();
                } else {
                    this.state.currentLevelData = this.gameData[this.state.currentDifficulty][this.state.currentLevelIndex];
                }
                this._renderLevel();
            },
            /**
             * Renders the UI for the current level.
             */
            _renderLevel() {
                this.elements.wordsColumn.innerHTML = '';
                this.elements.meaningsColumn.innerHTML = '';
                this._updateFeedback('', false);
                this.elements.nextLevelBtn.classList.remove('show');
                const createMatchItem = (itemData) => {
                    const item = document.createElement('div');
                    item.className = 'match-item';
                    item.textContent = itemData.text;
                    item.dataset.id = itemData.id;
                    return item;
                };
                const words = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.word, id: item.id })));
                const meanings = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.meaning, id: item.id })));
                words.forEach(item => this.elements.wordsColumn.appendChild(createMatchItem(item)));
                meanings.forEach(item => this.elements.meaningsColumn.appendChild(createMatchItem(item)));
            },
            _handleDifficultyChange(event) {
                const target = event.target.closest('.difficulty-btn');
                if (!target) return;
                this.state.currentDifficulty = target.dataset.difficulty;
                this.state.currentLevelIndex = 0;
                document.querySelectorAll('.difficulty-btn').forEach(btn => btn.classList.remove('active'));
                target.classList.add('active');
                this._loadLevel();
            },
            _handleItemClick(event) {
                const item = event.target.closest('.match-item');
                if (!item || item.classList.contains('correct')) return;
                if (!this.state.selectedItem) {
                    this.state.selectedItem = item;
                    item.classList.add('selected');
                } else {
                    if (this.state.selectedItem === item || this.state.selectedItem.parentElement === item.parentElement) {
                        this.state.selectedItem.classList.remove('selected');
                        this.state.selectedItem = null;
                        return;
                    }
                    if (this.state.selectedItem.dataset.id === item.dataset.id) { // Match
                        const matchedId = parseInt(item.dataset.id);
                        this.state.wordBank.get(matchedId).lastSeen = Date.now();
                        this.state.selectedItem.classList.add('correct');
                        item.classList.add('correct');
                        this.state.matchesFound++;
                        // Clear selection and check for level completion
                        this.state.selectedItem = null;
                        if (this.state.matchesFound === this.state.currentLevelData.length) {
                            this._updateFeedback('Round Complete!', true);
                            this.elements.nextLevelBtn.classList.add('show');
                        }
                    } else { // No match
                        this.state.selectedItem.classList.add('incorrect');
                        item.classList.add('incorrect');
                        const selected = this.state.selectedItem; // store temporarily
                        setTimeout(() => {
                            selected.classList.remove('incorrect', 'selected');
                            item.classList.remove('incorrect');
                        }, 500);
                        this.state.selectedItem = null;
                    }
                }
            },
            _handleNextLevel() {
                if (this.state.currentDifficulty !== 'review') {
                    const levels = this.gameData[this.state.currentDifficulty].length;
                    this.state.currentLevelIndex = (this.state.currentLevelIndex + 1) % levels;
                }
                this._loadLevel();
            },
            _updateFeedback(message, isSuccess) {
                this.elements.feedback.textContent = message;
                this.elements.feedback.style.color = isSuccess ? 'var(--success-color)' : 'var(--error-color)';
            },
            _generateSmartReviewData() {
                const seenWords = Array.from(this.state.wordBank.values()).filter(word => word.lastSeen > 0);
                const unseenWords = Array.from(this.state.wordBank.values()).filter(word => word.lastSeen === 0);
                // Prioritize words that have been seen but not recently.
                seenWords.sort((a, b) => a.lastSeen - b.lastSeen); 
                // Combine lists: oldest seen words first, then unseen words.
                const reviewPool = [...seenWords, ...unseenWords];
                return reviewPool.slice(0, 3);
            },
            _shuffle: (array) => array.sort(() => Math.random() - 0.5),
            // --- Menu and Modal Methods ---

                this._renderMenu();
                this.elements.menu.classList.toggle('show');

                        <textarea placeholder="Write notes here..."></textarea>
                    </div>`;
                document.getElementById('archive-btn-dynamic').addEventListener('click', this._openArchive.bind(this));
            },
            _openArchive() {
                this.elements.archiveList.innerHTML = '';
                this.elements.menu.classList.remove('show');
                const seenWords = Array.from(this.state.wordBank.values()).filter(word => word.lastSeen > 0);
                if (seenWords.length === 0) {
                    this.elements.archiveList.innerHTML = '<li>No words practiced yet. Complete a match!</li>';
                } else {
                    seenWords.sort((a, b) => b.lastSeen - a.lastSeen); // Most recent first
                    seenWords.forEach(item => {
                        const li = document.createElement('li');
                        li.innerHTML = `<strong>${item.word}</strong> = <em>${item.meaning}</em>`;
                        this.elements.archiveList.appendChild(li);
                    });
                }
                this.elements.archiveModal.style.display = 'block';
            },
            _closeArchive() {
                this.elements.archiveModal.style.display = 'none';
            }
        };
        // Start the application once the DOM is fully loaded.
        document.addEventListener('DOMContentLoaded', () => {
            LingoFlow.init();
        });

                };
                const masterList = Object.values(this.gameData).flat(2);
                masterList.forEach(word => this.state.wordBank.set(word.id, {...word, lastSeen: 0}));
                this._bindEvents();
                this._loadAdaptiveLevel();
            },
            _bindEvents() {
                this.elements.difficultySelector.addEventListener('click', this._handleDifficultyChange.bind(this));
                this.elements.nextLevelBtn.addEventListener('click', this._loadAdaptiveLevel.bind(this));
                this.elements.menuIcon.addEventListener('click', this._toggleMenu.bind(this));
                this.elements.closeArchiveBtn.addEventListener('click', this._closeArchive.bind(this));
                this.elements.wordsColumn.addEventListener('click', this._handleItemClick.bind(this));
                this.elements.meaningsColumn.addEventListener('click', this._handleItemClick.bind(this));
                window.addEventListener('click', (event) => {
                    if (event.target === this.elements.archiveModal) this._closeArchive();
                    if (this.elements.menu.classList.contains('show') && !this.elements.menu.contains(event.target) && !this.elements.menuIcon.contains(event.target)) {
                        this.elements.menu.classList.remove('show');
                    }
                });

            /**
             * Generates and loads an adaptive level using SRS logic on the current difficulty's word pool.
             */
            _loadAdaptiveLevel() {
                this.state.matchesFound = 0;
                this.state.selectedItem = null;
                // 1. Get the pool of words for the current difficulty.
                const wordPoolIds = new Set(Object.values(this.gameData[this.state.currentDifficulty]).flat().map(w => w.id));
                const difficultyWordBank = Array.from(this.state.wordBank.values()).filter(word => wordPoolIds.has(word.id));
                // 2. Separate into words the user hasn't seen and words they have.
                const seenWords = difficultyWordBank.filter(word => word.lastSeen > 0);
                const unseenWords = difficultyWordBank.filter(word => word.lastSeen === 0);
                // 3. Sort the seen words by last-seen time (oldest first).
                seenWords.sort((a, b) => a.lastSeen - b.lastSeen); 
                // 4. Create the final pool: oldest seen words, followed by any unseen words.
                const reviewPool = [...seenWords, ...unseenWords];
                // 5. Select the top 3 words most in need of review.
                this.state.currentLevelData = reviewPool.slice(0, 3);
                this._renderLevel();
            },
            /**
             * Renders the UI for the current level.
             */
            _renderLevel() {
                this.elements.wordsColumn.innerHTML = '';
                this.elements.meaningsColumn.innerHTML = '';
                this._updateFeedback('');
                this.elements.nextLevelBtn.classList.remove('show');
                if(this.state.currentLevelData.length === 0) {
                    this._updateFeedback(`All words in ${this.state.currentDifficulty} learned!`);
                    return;
                }
                const createMatchItem = (itemData) => {
                    const item = document.createElement('div');
                    item.className = 'match-item';
                    item.textContent = itemData.text;
                    item.dataset.id = itemData.id;
                    return item;
                };
                const words = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.word, id: item.id })));
                const meanings = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.meaning, id: item.id })));
                words.forEach(item => this.elements.wordsColumn.appendChild(createMatchItem(item)));
                meanings.forEach(item => this.elements.meaningsColumn.appendChild(createMatchItem(item)));
            },
            _handleDifficultyChange(event) {
                const target = event.target.closest('.difficulty-btn');
                if (!target) return;
                this.state.currentDifficulty = target.dataset.difficulty;
                document.querySelectorAll('.difficulty-btn').forEach(btn => btn.classList.remove('active'));
                target.classList.add('active');
                this._loadAdaptiveLevel();
            },
            _handleItemClick(event) {
                const item = event.target.closest('.match-item');
                if (!item || item.classList.contains('correct')) return;
                if (!this.state.selectedItem) {
                    this.state.selectedItem = item;
                    item.classList.add('selected');
                } else {
                    if (this.state.selectedItem === item || this.state.selectedItem.parentElement === item.parentElement) {
                        this.state.selectedItem.classList.remove('selected');
                        this.state.selectedItem = null;
                        return;
                    }
                    if (this.state.selectedItem.dataset.id === item.dataset.id) { // Match
                        const matchedId = parseInt(item.dataset.id, 10);
                        // This is the core of SRS: update the timestamp on success.
                        this.state.wordBank.get(matchedId).lastSeen = Date.now();
                        this.state.selectedItem.classList.add('correct');
                        item.classList.add('correct');
                        this.state.matchesFound++;
                        this.state.selectedItem = null;
                        if (this.state.matchesFound === this.state.currentLevelData.length) {
                            this._updateFeedback('Round Complete!', true);
                            this.elements.nextLevelBtn.classList.add('show');
                        }
                    } else { // No match
                        this.state.selectedItem.classList.add('incorrect');
                        item.classList.add('incorrect');
                        const selected = this.state.selectedItem;
                        setTimeout(() => {
                            selected.classList.remove('incorrect', 'selected');
                            item.classList.remove('incorrect');
                        }, 500);
                        this.state.selectedItem = null;
                    }
                }
            },
            _updateFeedback(message, isSuccess = false) {
                this.elements.feedback.textContent = message;
                this.elements.feedback.style.color = isSuccess ? 'var(--success-color)' : 'var(--light-text-color)';
            },

                this.elements.archiveList.innerHTML = '';
                this.elements.menu.classList.remove('show');
                const seenWords = Array.from(this.state.wordBank.values()).filter(word => word.lastSeen > 0);
                if (seenWords.length === 0) {
                    this.elements.archiveList.innerHTML = '<li>No words practiced yet. Complete a match!</li>';
                } else {
                    seenWords.sort((a, b) => b.lastSeen - a.lastSeen); // Most recent first
                    seenWords.forEach(item => {
                        const li = document.createElement('li');
                        li.innerHTML = `<strong>${item.word}</strong> = <em>${item.meaning}</em>`;
                        this.elements.archiveList.appendChild(li);
                    });
                }
                this.elements.archiveModal.style.display = 'block';
            },
            _closeArchive() {
                this.elements.archiveModal.style.display = 'none';
            }

                this.elements.wordsColumn.addEventListener('click', this._handleItemClick.bind(this));
                this.elements.meaningsColumn.addEventListener('click', this._handleItemClick.bind(this));
                window.addEventListener('click', (event) => {
                    if (event.target === this.elements.archiveModal) this._closeArchive();
                    if (this.elements.menu.classList.contains('show') && !this.elements.menu.contains(event.target) && !this.elements.menuIcon.contains(event.target)) {
                        this.elements.menu.classList.remove('show');
                    }
                });
            },
            _loadAdaptiveLevel() {
                this.state.matchesFound = 0;
                this.state.selectedItem = null;
                const wordPoolIds = new Set(this.gameData[this.state.currentDifficulty].map(w => w.id));
                const difficultyWordBank = Array.from(this.state.wordBank.values()).filter(word => wordPoolIds.has(word.id));
                const seenWords = difficultyWordBank.filter(word => word.lastSeen > 0);
                const unseenWords = difficultyWordBank.filter(word => word.lastSeen === 0);
                seenWords.sort((a, b) => a.lastSeen - b.lastSeen); 
                const reviewPool = [...seenWords, ...unseenWords];
                this.state.currentLevelData = reviewPool.slice(0, 3);
                this._renderLevel();
            },
            _renderLevel() {
                this.elements.wordsColumn.innerHTML = '';
                this.elements.meaningsColumn.innerHTML = '';
                this._updateFeedback('');
                this.elements.nextLevelBtn.classList.remove('show');
                if(this.state.currentLevelData.length === 0) {
                    this._updateFeedback(`All words in ${this.state.currentDifficulty} learned!`);
                    return;
                }
                const createMatchItem = (itemData) => {
                    const item = document.createElement('div');
                    item.className = 'match-item';
                    item.textContent = itemData.text;
                    item.dataset.id = itemData.id;
                    return item;
                };
                const words = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.word, id: item.id })));
                const meanings = this._shuffle(this.state.currentLevelData.map(item => ({ text: item.meaning, id: item.id })));
                words.forEach(item => this.elements.wordsColumn.appendChild(createMatchItem(item)));
                meanings.forEach(item => this.elements.meaningsColumn.appendChild(createMatchItem(item)));
            },
            _handleDifficultyChange(event) {
                const target = event.target.closest('.difficulty-btn');
                if (!target) return;
                this.state.currentDifficulty = target.dataset.difficulty;
                document.querySelectorAll('.difficulty-btn').forEach(btn => btn.classList.remove('active'));
                target.classList.add('active');
                this._loadAdaptiveLevel();
            },
            _handleItemClick(event) {
                const item = event.target.closest('.match-item');
                if (!item || item.classList.contains('correct')) return;
                if (!this.state.selectedItem) {
                    this.state.selectedItem = item;
                    item.classList.add('selected');
                } else {
                    if (this.state.selectedItem === item || this.state.selectedItem.parentElement === item.parentElement) {
                        this.state.selectedItem.classList.remove('selected');
                        this.state.selectedItem = null;
                        return;
                    }
                    if (this.state.selectedItem.dataset.id === item.dataset.id) {
                        const matchedId = parseInt(item.dataset.id, 10);
                        this.state.wordBank.get(matchedId).lastSeen = Date.now();
                        this.state.selectedItem.classList.add('correct');
                        item.classList.add('correct');
                        this.state.matchesFound++;
                        this.state.selectedItem = null;
                        if (this.state.matchesFound === this.state.currentLevelData.length) {
                            this._updateFeedback('Round Complete!', true);
                            this.elements.nextLevelBtn.classList.add('show');
                        }
                    } else {
                        this.state.selectedItem.classList.add('incorrect');
                        item.classList.add('incorrect');
                        const selected = this.state.selectedItem;
                        setTimeout(() => {
                            selected.classList.remove('incorrect', 'selected');
                            item.classList.remove('incorrect');
                        }, 500);
                        this.state.selectedItem = null;
                    }
                }
            },
            _updateFeedback(message, isSuccess = false) {
                this.elements.feedback.textContent = message;
                this.elements.feedback.style.color = isSuccess ? 'var(--success-color)' : 'var(--light-text-color)';
            },
            _shuffle: (array) => array.sort(() => Math.random() - 0.5),
            _toggleMenu() {
                this._renderMenu();
                this.elements.menu.classList.toggle('show');
            },
            _renderMenu() {
                this.elements.menu.innerHTML = `
                    <div class="tool-card">
                        <h3>Word Archive</h3>
                        <p>Review all words you've practiced.</p>
                        <button id="archive-btn-dynamic">View Archive</button>
                    </div>
                    <div class="tool-card">
                        <h3>Quick Notes</h3>
                        <textarea placeholder="Write notes here..."></textarea>
                    </div>`;
                document.getElementById('archive-btn-dynamic').addEventListener('click', this._openArchive.bind(this));
            },
            _openArchive() {
                this.elements.archiveList.innerHTML = '';
                this.elements.menu.classList.remove('show');
                const seenWords = Array.from(this.state.wordBank.values()).filter(word => word.lastSeen > 0);
                if (seenWords.length === 0) {
                    this.elements.archiveList.innerHTML = '<li>No words practiced yet. Complete a match!</li>';
                } else {
                    seenWords.sort((a, b) => b.lastSeen - a.lastSeen);
                    seenWords.forEach(item => {
                        const li = document.createElement('li');
                        li.innerHTML = `<strong>${item.word}</strong> = <em>${item.meaning}</em>`;
                        this.elements.archiveList.appendChild(li);
                    });
                }
                this.elements.archiveModal.style.display = 'block';
            },
            _closeArchive() {
                this.elements.archiveModal.style.display = 'none';
            }
        };
        document.addEventListener('DOMContentLoaded', () => LingoFlow.init());
    </script>
